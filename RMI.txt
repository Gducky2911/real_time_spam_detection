        ByteService sv = (ByteService) rg.lookup("RMIByteService");
        byte[] a = sv.requestData("B21DCCN900", "Ta11vtlf");
        for(byte x: a) System.out.print(x + " ");
        System.out.println("");
        //b. Đếm

        for (byte x : a) cnt[x]++;
        byte ptuMin = a[0];
        int slMin = cnt[a[0]];

            if (cnt[x] < slMin) {
                ptuMin = x;
                slMin = cnt[x];
            }
        }
        System.out.println(ptuMin + " " + slMin);
        // c. Gửi
        byte[] res = {ptuMin, (byte)slMin};
        sv.submitData("B21DCCN900", "Ta11vtlf", res);
    }
}

/**
 * ĐỈNH CỤC BỘ
Một chương trình RMI Server cung cấp giao diện DataService trong package RMI, được đăng ký với RegistryServer bằng tên RMIDataService.

Yêu cầu
a. Sử dụng mã sinh viên và mã câu hỏi làm tham số khi gọi RMI. Ví dụ: studentCode="B21DCCN001", qCode="A1B2C3D4".

b. Triệu gọi requestData(studentCode, qCode) để nhận một mảng số nguyên từ server. Ví dụ: [3,8,5,10,7].

c. Một phần tử là đỉnh cục bộ nếu lớn hơn phần tử liền kề bên trái và bên phải; phần tử đầu/cuối chỉ cần lớn hơn phần tử kề duy nhất. Trả về vị trí theo chỉ số bắt đầu từ 1.

d. Triệu gọi submitData(studentCode, qCode, data) để gửi List<Integer> các vị trí đỉnh cục bộ. Ví dụ: [2,4].

e. Kết thúc chương trình client.
 */


import java.rmi.registry.LocateRegistry;
import java.rmi.registry.Registry;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

public class DinhCucBo {

    public static void main(String[] args) throws Exception {

        String studentCode = "B21DCCN900";
        String qCode = "xJwA0y3m";

        // a. Kết nối đến RMI Registry
        Registry rg = LocateRegistry.getRegistry(
                "36.50.135.242",
                1099
        );

        // Tìm dịch vụ có tên RMIDataService
        DataService sv =
                (DataService) rg.lookup("RMIDataService");

        // b. Nhận dữ liệu
        // requestData trả về Object nên phải ép kiểu sang int[]
        int[] a = (int[]) sv.requestData(studentCode, qCode);

        System.out.println("Mang nhan duoc: "
                + Arrays.toString(a));

        // Danh sách chứa vị trí các đỉnh cục bộ
        List<Integer> res = new ArrayList<>();

        int n = a.length;

        // Trường hợp mảng chỉ có một phần tử
        if (n == 1) {
            res.add(1);
        } else {

            // Kiểm tra phần tử đầu tiên
            if (a[0] > a[1]) {
                res.add(1);
            }

            // Kiểm tra các phần tử ở giữa
            for (int i = 1; i < n - 1; i++) {
                if (a[i] > a[i - 1]
                        && a[i] > a[i + 1]) {

                    // Đề yêu cầu vị trí bắt đầu từ 1
                    res.add(i + 1);
                }
            }

            // Kiểm tra phần tử cuối cùng
            if (a[n - 1] > a[n - 2]) {
                res.add(n);
            }
        }

        System.out.println("Vi tri dinh cuc bo: " + res);

        // d. Gửi kết quả về Server
        sv.submitData(studentCode, qCode, res);

        System.out.println("Da gui ket qua ve Server.");
    }
}

 Một chương trình RMI Server cung cấp giao diện DataService trong package RMI, được đăng ký với RegistryServer bằng tên RMIDataService.

Yêu cầu

b. Triệu gọi requestData(studentCode, qCode) để nhận chuỗi CSV các số thực. Ví dụ: 10.00,20.00,30.00,40.00.

c. Tính giá trị trung bình, độ lệch chuẩn trên toàn bộ dãy và phân vị 95 tại vị trí ceil(n * 0.95) - 1 sau khi sắp xếp tăng dần.

d. Triệu gọi submitData(studentCode, qCode, data) để gửi chuỗi average=<avg>;stddev=<stddev>;p95=<p95>, các số thực làm tròn 02 chữ số thập phân. Ví dụ: average=25.00;stddev=11.18;p95=40.00.

e. Kết thúc chương trình client.
 */

import java.rmi.registry.LocateRegistry;
import java.rmi.registry.Registry;
import java.util.Arrays;
import java.util.Locale;

public class SoThucNangCao {

    public static void main(String[] args) throws Exception {

        String studentCode = "B21DCCN900";
        String qCode = "M6gWmdYO";

        // a. Kết nối tới RMI Registry
        Registry rg = LocateRegistry.getRegistry(
                "36.50.135.242",
                1099
        );

        DataService sv =
                (DataService) rg.lookup("RMIDataService");

        // b. Nhận chuỗi CSV từ Server
        String duLieu = (String) sv.requestData(
                studentCode,
                qCode
        );

        System.out.println("Chuoi nhan duoc: " + duLieu);

        // Tách chuỗi theo dấu phẩy
        String[] cacPhanTu = duLieu.split(",");

        // Tạo mảng số thực
        double[] a = new double[cacPhanTu.length];

        for (int i = 0; i < cacPhanTu.length; i++) {
            a[i] = Double.parseDouble(cacPhanTu[i].trim());
        }

        int n = a.length;

        // c1. Tính trung bình
        double tong = 0;

        for (int i = 0; i < n; i++) {
            tong += a[i];
        }

        double trungBinh = tong / n;

        // c2. Tính độ lệch chuẩn của toàn bộ dãy
        double tongBinhPhuongDoLech = 0;

        for (int i = 0; i < n; i++) {
            double doLech = a[i] - trungBinh;

            tongBinhPhuongDoLech += doLech * doLech;
        }

        double doLechChuan =
                Math.sqrt(tongBinhPhuongDoLech / n);

        // c3. Sắp xếp tăng dần để tìm phân vị 95
        Arrays.sort(a);

        int viTriP95 = (int) Math.ceil(n * 0.95) - 1;

        double p95 = a[viTriP95];

        System.out.println("Trung binh: " + trungBinh);
        System.out.println("Do lech chuan: " + doLechChuan);
        System.out.println("P95: " + p95);

        // d. Tạo chuỗi kết quả, làm tròn 2 chữ số thập phân
        String ketQua = String.format(
                Locale.US,
                "average=%.2f;stddev=%.2f;p95=%.2f",
                trungBinh,
                doLechChuan,
                p95
        );

        System.out.println("Ket qua gui ve: " + ketQua);

        // Gửi kết quả về Server
        sv.submitData(
                studentCode,
                qCode,
                ketQua
        );
    }
}

b. Xác định tất cả các bộ ba số nguyên (a, b, c) sao cho a² + b² = c² và a < b < c ≤ N. Kết quả trả về
là danh sách các bộ ba số Pythagore thỏa mãn yêu cầu.
Ví dụ: Với N = 20, kết quả là [(3, 4, 5), (5, 12, 13), (8, 15, 17)].

import java.rmi.*;
import java.rmi.registry.*;
import java.util.*;
import RMI.DataService;
public class BoBaPytago {
    public static void main(String[] args) throws Exception {
        // a. Nhận dữ liệu
        Registry rg = LocateRegistry.getRegistry("36.50.135.242", 1099);
        DataService sv = (DataService) rg.lookup("RMIDataService");
        int N = (int) sv.requestData("B21DCCN900", "xJwA0y3m");
        System.out.println(N);
        // b. Xử lý: Tìm danh sách các bộ ba số Pythagore
        List<List<Integer>> res = new ArrayList<>();
        for (int a = 1; a <= N; a++) {
            for (int b = a + 1; b <= N; b++) {
                for (int c = b + 1; c <= N; c++) {
                    if (a * a + b * b == c * c) {
                        List<Integer> triple = new ArrayList<>();
                        triple.add(a); triple.add(b); triple.add(c);
                        res.add(triple);
                    }
                }
            }
        }
        System.out.println(res);
        // c. Gửi kết quả
        sv.submitData("B21DCCN900", "xJwA0y3m", res);
    }
}

import java.rmi.*;
import java.rmi.registry.*;
import java.util.*;
import RMI.DataService;
public class HoanViTiepTheo {
    public static void main(String[] args) throws Exception {
        // a. Nhận dữ liệu
        Registry rg = LocateRegistry.getRegistry("203.162.10.109", 1099);
        DataService sv = (DataService) rg.lookup("RMIDataService");
        String s = (String) sv.requestData("B21DCCN564", "3Qe0Qb5w");
        System.out.println(s);
        //Đẩy hết vào mảng để xử lý
        s = s.replace(",", " ");
        String[] tmp = s.trim().split("\\s+");
        ArrayList<Integer> a = new ArrayList<>();
        for (String x : tmp) a.add(Integer.parseInt(x));
        int n = a.size();
        //b. Sinh hoán vị kế tiếp
        // Bước 1: Tìm vị trí đầu tiên từ cuối mà tại đó nó nhỏ hơn số đằng sau nó
        int i = n - 2;
        while (i >= 0 && a.get(i) >= a.get(i + 1)) i--;
        // Bước 2: Kiểm tra nếu không còn hoán vị nào lớn hơn
        if (i < 0) Collections.sort(a); // Đưa về hoán vị đầu tiên
        else {
            // Bước 3: Tìm vị trí đầu tiên từ cuối lớn hơn a[i]
            int j = n - 1;
            while (a.get(j) <= a.get(i)) j--;
            Collections.swap(a, i, j);// Hoán đổi a[i] và a[j]
            // Bước 4: Lật ngược đoạn từ i + 1 đến cuối mảng
            int l = i + 1, r = n - 1;
            while (l < r) {
                Collections.swap(a, l, r);
                l++;
                r--;
            }
        }
        String res = "";
        for(int x: a) res+=String.format("%d,", x);
        res = res.substring(0, res.length() - 1);
        System.out.println(res);
        // c. Gửi kết quả
        sv.submitData("B21DCCN564", "3Qe0Qb5w", res);
    }
}

        else if (c=='X') return 10;
        else if (c=='L') return 50;
        else if (c=='C') return 100;
        else if (c=='D') return 500;

        tong+= trans(s .charAt(s.length()-1));

package RMI.B21DCCN053;
import RMI.DataService;
import java.rmi.registry.LocateRegistry;
import java.rmi.registry.Registry;
import java.util.*;
public class SinhToHop {
    static int ok; 
    public static void kt(int a[], int k) {
        for (int i = 1; i <= k; i++) a[i] = i;
    }
    public static void sinh(int a[], int n, int k) {
        int i = k;
        while (i >= 1 && a[i] == n - k + i) i--;
        if (i == 0) {
            ok = 0;
        } else {
            a[i]++;
            for (int j = i + 1; j <= k; j++) a[j] = a[j - 1] + 1;
        }
    }
    public static void main(String[] args) throws Exception{
        Registry rg = LocateRegistry.getRegistry("203.162.10.109", 1099);
        DataService sv = (DataService) rg.lookup("RMIDataService");
        String s = (String)sv.requestData("B21DCCN053", "juC3u7C6");
        //String s = "2, 0, 5, 8, 1 ;3"; 
        System.out.println(s);
        int idx = s.indexOf(";");
        String s1 = s.substring(0, idx), s2 = s.substring(idx + 1);
        int k = Integer.parseInt(s2.trim());// Lấy số k
        // Lấy ra mảng và sắp xếp
        ArrayList<Integer> a = new ArrayList<>();
        s1 = s1.replace(",", " ");
        String[] tmp = s1.trim().split("\\s+");
        for (String x : tmp) a.add(Integer.parseInt(x));
        // Bắt đầu sinh tổ hợp. Khởi tạo
        int n = a.size();
        int[] genIdx = new int[k + 1];
        kt(genIdx, k);
        ok = 1;
        // Duyệt qua từng tổ hợp chỉ số
        ArrayList<ArrayList<Integer>> ans = new ArrayList<>();
        while (ok == 1) {
            ArrayList<Integer> tmp1 = new ArrayList<>();
            for (int i = 1; i <= k; i++) tmp1.add(a.get(genIdx[i] - 1));
            ans.add(tmp1);
            sinh(genIdx, n, k);
        }
        //c.
        System.out.println(ans);
        sv.submitData("B21DCCN053", "juC3u7C6", ans);
    }
}
